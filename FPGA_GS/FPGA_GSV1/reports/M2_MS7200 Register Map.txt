Note:  
1. 
MS7200 registers not in the following list should not be written. 
2. 
MS7200 HDMI controller should be accessed through i2c burst mode. 
Example:  
VOID mculib_i2c_burstread _16bidx8bval(UINT8 u8_address, UINT16 u16_index, UINT16 u16_length, UINT8 *pu8_value) 
{ 
UINT16 i; 
_i2c_start(); 
if (!_i2c_write_byte(u8_address)) 
goto STOP; 
if (!_i2c_write_byte( (UINT8)(u16_index) )) 
goto STOP; 
if (!_i2c_write_byte( (UINT8)(u16_index >> 8) )) 
goto STOP; 
_i2c_start(); 
if (!_i2c_write_byte(u8_address | 0x01)) 
goto STOP; 
for (i = 0; i < (u16_length-1); i ++) 
{ 
*pu8_value++ = _i2c_read_byte(FALSE); 
} 
*pu8_value = _i2c_read_byte(TRUE); 
_i2c_stop(); 
}
